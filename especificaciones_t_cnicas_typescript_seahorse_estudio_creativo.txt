/**
 * Las 5 disciplinas exclusivas de autor definidas para la marca
 */
export type ProductCategory = 
  | 'cuadros'        // Acrílicos y técnicas mixtas sobre lino crudo
  | 'ropa'           // Prendas de autor con bordados y tintes botánicos
  | 'peluches'       // Figuras textiles coleccionables rellenas a mano
  | 'resina'         // Posavasos, aretes, dijes y encapsulados botánicos
  | 'cemento_blanco'; // Floreros escultóricos, bandejas onduladas y portavelas

/**
 * Estados de disponibilidad comercial y operativa
 */
export type ProductAvailability = 
  | 'disponible'     // Entrega inmediata (en stock)
  | 'bajo_pedido'    // Requiere tiempo de fraguado o confección
  | 'personalizable' // Permite co-creación a medida
  | 'agotado';       // Sin existencias inmediatas

export interface ProductDimensions {
  readonly widthCm?: number;
  readonly heightCm?: number;
  readonly depthCm?: number;
  readonly diameterCm?: number;
  readonly weightGrams?: number;
}

export interface ProductPrice {
  /** Valor monetario en Pesos Colombianos (COP) */
  readonly amountCop: number;
  /** Representación formateada con separador de miles (ej: "$65.000 COP") */
  readonly formattedCop: string;
  readonly isStartingPrice?: boolean;
}

export interface ProductItem {
  readonly id: string;
  readonly slug: string;
  readonly name: string;
  readonly subtitle?: string;
  readonly category: ProductCategory;
  readonly availability: ProductAvailability;
  readonly price: ProductPrice;
  readonly materials: readonly string[];
  readonly curingDaysEstimate?: number; // Días de reposo/fraguado en taller
  readonly description: string;
  readonly editorialNotes?: string;
  readonly primaryImageUrl: string;
  readonly galleryImageUrls: readonly string[];
  readonly dimensions?: ProductDimensions;
  readonly isLimitedEdition?: boolean;
  readonly editionTotalCount?: number;
  readonly editionCurrentNumber?: number;
  readonly createdAt: string; // ISO 8601
  readonly updatedAt: string; // ISO 8601
}

// ============================================================================
// 4. CARRITO, CHECKOUT Y LOGÍSTICA EN COLOMBIA
// ============================================================================

export interface CartLineItem {
  readonly productId: string;
  readonly productName: string;
  readonly category: ProductCategory;
  readonly unitPriceCop: number;
  readonly quantity: number;
  readonly curingDaysEstimate?: number;
  readonly primaryImageUrl: string;
  readonly subtotalCop: number;
}

export type ColombianCarrier = 
  | 'coordinadora'
  | 'servientrega'
  | 'inter_rapidisimo'
  | 'envia';

export interface ColombianAddress {
  /** Destinatario con documento de identidad para guías oficiales */
  readonly recipientFullName: string;
  readonly nationalIdNumber: string; // Cédula de Ciudadanía o NIT
  readonly phoneOrWhatsapp: string;
  /** Nomenclatura urbana: Carrera, Calle, Avenida, Diagonal, Manzana, etc. */
  readonly streetAddress: string;
  readonly apartmentOrSuite?: string;
  readonly neighborhood?: string; // Barrio
  readonly city: string; // Municipio o Distrito Capital
  readonly department: string; // Departamento de Colombia (32 + Bogotá D.C.)
  readonly postalCode?: string;
}

export type ColombianPaymentGateway = 
  | 'mercado_pago'  // Tarjetas crédito/débito y PSE hacia todos los bancos
  | 'nequi'         // Notificación Push a app móvil o Código QR
  | 'daviplata';    // Transferencia simplificada con número celular

export type PaymentStatus = 
  | 'pendiente'
  | 'procesando'
  | 'aprobado'
  | 'rechazado'
  | 'reembolsado';

export interface FinancialBreakdown {
  readonly itemsSubtotalCop: number;
  readonly curatedPackagingFeeCop: number; // Empaque especial artesanal
  readonly domesticShippingFeeCop: number;  // Flete nacional cotizado
  readonly totalCop: number;
}

export interface Order {
  readonly orderId: string; // Formato ej: "#SH-2024-1089"
  readonly items: readonly CartLineItem[];
  readonly shippingAddress: ColombianAddress;
  readonly selectedCarrier: ColombianCarrier;
  readonly carrierTrackingNumber?: string;
  readonly paymentGateway: ColombianPaymentGateway;
  readonly paymentStatus: PaymentStatus;
  readonly financialSummary: FinancialBreakdown;
  readonly orderStatus: 
    | 'pago_confirmado'
    | 'en_taller_secado'
    | 'listo_para_despacho'
    | 'en_transito_nacional'
    | 'entregado'
    | 'novedad_logistica';
  readonly specialNotes?: string;
  readonly placedAt: string;
  readonly shippedAt?: string;
  readonly deliveredAt?: string;
}

// ============================================================================
// 6. CONTRATOS DE SERVICIOS Y REPOSITORIOS (API CONTRACTS)
// ============================================================================

export interface ProductCatalogService {
  listProducts(filters?: {
    category?: ProductCategory;
    availability?: ProductAvailability;
    searchQuery?: string;
    maxPriceCop?: number;
    sort?: 'newest' | 'price_asc' | 'price_desc';
  }): Promise<readonly ProductItem[]>;

  getProductBySlug(slug: string): Promise<ProductItem | null>;
}

export interface CustomizationService {
  submitCustomRequest(payload: Omit<CustomPieceRequest, 'requestId' | 'status' | 'createdAt'>): Promise<{
    requestId: string;
    whatsappDirectUrl: string;
  }>;
}

export interface CheckoutAndLogisticsService {
  calculateShippingFee(params: {
    department: string;
    city: string;
    carrier: ColombianCarrier;
    packageWeightGrams: number;
  }): Promise<{ feeCop: number; estimatedDeliveryDays: string }>;

  createOrder(payload: {
    items: readonly CartLineItem[];
    address: ColombianAddress;
    carrier: ColombianCarrier;
    paymentGateway: ColombianPaymentGateway;
  }): Promise<Order>;

  verifyPaymentStatus(orderId: string): Promise<{
    paymentStatus: PaymentStatus;
    transactionReference?: string;
  }>;
}

export interface AdminManagementService {
  getOperationalKPIs(): Promise<AdminKPIs>;
  getOrdersList(filter?: {
    carrier?: ColombianCarrier;
    orderStatus?: Order['orderStatus'];
    city?: string;
  }): Promise<readonly Order[]>;
  generateDianManifest(dateIso: string): Promise<{ downloadUrl: string; manifestId: string }>;
  getSecurityLogs(limit?: number): Promise<readonly SecurityAuditEvent[]>;
}
